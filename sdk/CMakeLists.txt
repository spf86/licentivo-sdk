cmake_minimum_required(VERSION 3.18)
project(licentivo_sdk VERSION 0.11.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_library(licentivo_c STATIC c/licentivo.c c/vendor/cJSON.c c/vendor/monocypher.c c/vendor/monocypher-ed25519.c)
add_library(Licentivo::C ALIAS licentivo_c)
target_include_directories(licentivo_c PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/c> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/licentivo/c>)
set_target_properties(licentivo_c PROPERTIES EXPORT_NAME C)
target_link_libraries(licentivo_c PUBLIC CURL::libcurl Threads::Threads)
if(WIN32)
  target_link_libraries(licentivo_c PUBLIC bcrypt advapi32)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(licentivo_c PUBLIC OpenSSL::Crypto)
endif()
if(NOT WIN32)
  target_link_libraries(licentivo_c PUBLIC m)
endif()
add_library(licentivo_cpp INTERFACE)
add_library(Licentivo::CPP ALIAS licentivo_cpp)
target_include_directories(licentivo_cpp INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/cpp> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/licentivo/cpp>)
set_target_properties(licentivo_cpp PROPERTIES EXPORT_NAME CPP)
target_link_libraries(licentivo_cpp INTERFACE licentivo_c)
option(LICENTIVO_BUILD_EXAMPLES "Build the client example applications" ON)
option(LICENTIVO_BUILD_TESTS "Build the shared signature-vector checker" ON)
if(LICENTIVO_BUILD_EXAMPLES)
  add_executable(licentivo_c_demo c/demo.c)
  target_link_libraries(licentivo_c_demo PRIVATE licentivo_c)
  add_executable(licentivo_cpp_demo cpp/demo.cpp)
  target_link_libraries(licentivo_cpp_demo PRIVATE licentivo_cpp)
  add_executable(licentivo_c_app c/app.c)
  target_link_libraries(licentivo_c_app PRIVATE licentivo_c)
  add_executable(licentivo_cpp_app cpp/app.cpp)
  target_link_libraries(licentivo_cpp_app PRIVATE licentivo_cpp)
  add_executable(licentivo_c_quickstart c/quickstart.c)
  target_link_libraries(licentivo_c_quickstart PRIVATE Licentivo::C)
  add_executable(licentivo_cpp_quickstart cpp/quickstart.cpp)
  target_link_libraries(licentivo_cpp_quickstart PRIVATE Licentivo::CPP)
endif()
if(LICENTIVO_BUILD_TESTS)
  add_executable(licentivo_vectors c/test_vectors.c)
  target_link_libraries(licentivo_vectors PRIVATE licentivo_c)
endif()
install(TARGETS licentivo_c licentivo_cpp EXPORT LicentivoTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES c/licentivo.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/licentivo/c)
install(DIRECTORY c/vendor/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/licentivo/c/vendor FILES_MATCHING PATTERN "*.h")
install(FILES cpp/licentivo.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/licentivo/cpp)
install(EXPORT LicentivoTargets NAMESPACE Licentivo:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Licentivo)
configure_package_config_file(LicentivoConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/LicentivoConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Licentivo)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/LicentivoConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/LicentivoConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/LicentivoConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Licentivo)
install(FILES LICENSE NOTICE c/vendor/cJSON-LICENSE c/vendor/Monocypher-LICENCE.md DESTINATION licenses)
